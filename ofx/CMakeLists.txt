cmake_minimum_required(VERSION 3.16)
project(CorridorKeyLinuxOFX VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  message(FATAL_ERROR "This experimental bridge currently targets Linux x86-64.")
endif()
set(OFX_INCLUDE_DIR "/opt/resolve/Developer/OpenFX/OpenFX-1.4/include" CACHE PATH "Directory containing ofxImageEffect.h (Resolve SDK or AcademySoftwareFoundation/openfx/include)")
if(NOT EXISTS "${OFX_INCLUDE_DIR}/ofxImageEffect.h")
  message(FATAL_ERROR "Set -DOFX_INCLUDE_DIR=/path/to/openfx/include")
endif()
add_library(CorridorKeyLinux MODULE plugin.cpp)
target_include_directories(CorridorKeyLinux PRIVATE "${OFX_INCLUDE_DIR}")
target_compile_features(CorridorKeyLinux PRIVATE cxx_std_17)
target_compile_options(CorridorKeyLinux PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
target_link_options(CorridorKeyLinux PRIVATE -Wl,-z,defs -Wl,-z,relro -Wl,-z,now)
set(BUNDLE "CorridorKeyLinux.ofx.bundle/Contents/Linux-x86-64")
set(RESOURCES "CorridorKeyLinux.ofx.bundle/Contents/Resources")
foreach(NOTICE IN ITEMS ../LICENSE ../THIRD_PARTY_NOTICES.md OPENFX_LICENSE README.md)
  get_filename_component(NOTICE_NAME "${NOTICE}" NAME)
  configure_file("${NOTICE}" "${CMAKE_BINARY_DIR}/${RESOURCES}/${NOTICE_NAME}" COPYONLY)
  install(FILES "${NOTICE}" DESTINATION "${RESOURCES}")
endforeach()
set_target_properties(CorridorKeyLinux PROPERTIES PREFIX "" SUFFIX ".ofx" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${BUNDLE}")
install(TARGETS CorridorKeyLinux LIBRARY DESTINATION "${BUNDLE}")
include(CTest)
if(BUILD_TESTING)
  add_executable(ofx_host_test ../tests/ofx_host.cpp)
  target_include_directories(ofx_host_test PRIVATE "${OFX_INCLUDE_DIR}")
  target_compile_features(ofx_host_test PRIVATE cxx_std_17)
  target_link_libraries(ofx_host_test PRIVATE dl pthread)
  add_test(NAME ofx_host COMMAND ofx_host_test $<TARGET_FILE:CorridorKeyLinux>)
  set_tests_properties(ofx_host PROPERTIES TIMEOUT 30)
endif()
