cmake_minimum_required(VERSION 3.16)
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
  set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum supported macOS version")
  set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "Supported macOS architecture")
endif()
project(CorridorKeyOFX VERSION 0.1.0 LANGUAGES CXX)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
    message(FATAL_ERROR "The Linux OFX bridge currently targets x86-64.")
  endif()
  set(OFX_BUNDLE "CorridorKeyLinux.ofx.bundle")
  set(OFX_BINARY_DIR "${OFX_BUNDLE}/Contents/Linux-x86-64")
  set(OFX_BINARY_NAME "CorridorKeyLinux")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    message(FATAL_ERROR "The macOS OFX bridge currently targets Apple Silicon (arm64).")
  endif()
  set(OFX_BUNDLE "CorridorKeyMacOS.ofx.bundle")
  set(OFX_BINARY_DIR "${OFX_BUNDLE}/Contents/MacOS")
  set(OFX_BINARY_NAME "CorridorKeyMacOS")
else()
  message(FATAL_ERROR "The OFX bridge currently targets Linux x86-64 and macOS arm64.")
endif()

set(OFX_INCLUDE_DIR "/opt/resolve/Developer/OpenFX/OpenFX-1.4/include" CACHE PATH "Directory containing ofxImageEffect.h (Resolve SDK or AcademySoftwareFoundation/openfx/include)")
if(NOT EXISTS "${OFX_INCLUDE_DIR}/ofxImageEffect.h")
  message(FATAL_ERROR "Set -DOFX_INCLUDE_DIR=/path/to/openfx/include")
endif()

add_library(CorridorKeyPlugin MODULE plugin.cpp)
target_include_directories(CorridorKeyPlugin PRIVATE "${OFX_INCLUDE_DIR}")
target_compile_features(CorridorKeyPlugin PRIVATE cxx_std_17)
target_compile_options(CorridorKeyPlugin PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(CorridorKeyPlugin PRIVATE -Wl,-z,defs -Wl,-z,relro -Wl,-z,now)
endif()
set_target_properties(CorridorKeyPlugin PROPERTIES
  PREFIX ""
  SUFFIX ".ofx"
  OUTPUT_NAME "${OFX_BINARY_NAME}"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${OFX_BINARY_DIR}")

set(RESOURCES "${OFX_BUNDLE}/Contents/Resources")
foreach(NOTICE IN ITEMS ../LICENSE ../THIRD_PARTY_NOTICES.md OPENFX_LICENSE README.md)
  get_filename_component(NOTICE_NAME "${NOTICE}" NAME)
  configure_file("${NOTICE}" "${CMAKE_BINARY_DIR}/${RESOURCES}/${NOTICE_NAME}" COPYONLY)
  install(FILES "${NOTICE}" DESTINATION "${RESOURCES}")
endforeach()
install(TARGETS CorridorKeyPlugin LIBRARY DESTINATION "${OFX_BINARY_DIR}")

if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  configure_file("Info.plist.in" "${CMAKE_BINARY_DIR}/${OFX_BUNDLE}/Contents/Info.plist" @ONLY)
  install(FILES "${CMAKE_BINARY_DIR}/${OFX_BUNDLE}/Contents/Info.plist" DESTINATION "${OFX_BUNDLE}/Contents")
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(ofx_host_test ../tests/ofx_host.cpp)
  target_include_directories(ofx_host_test PRIVATE "${OFX_INCLUDE_DIR}")
  target_compile_features(ofx_host_test PRIVATE cxx_std_17)
  target_link_libraries(ofx_host_test PRIVATE "${CMAKE_DL_LIBS}" Threads::Threads)
  add_test(NAME ofx_host COMMAND ofx_host_test $<TARGET_FILE:CorridorKeyPlugin>)
  set_tests_properties(ofx_host PROPERTIES TIMEOUT 30)
endif()
